\begin{frame}[t]{Koornwinder polynomials: overview}
Koornwinder polynomials are Laurent polynomials in $n$ variables $z_1,\dots,z_n$, depending on six parameters $q,t,a,b,c,d$.
\begin{columns}[T]
\column{0.64\textwidth}
\begin{itemize}
  \item Koornwinder '92: \notion{symmetric} polynomials, indexed by partitions, introduced as multivariable orthogonal polynomials.
  \item Cherednik '95: introduced double affine Hecke algebras of reduced types.
  \item Sahi '99, '00: introduced the double affine Hecke algebra of type $(C^\vee,C)$ and used it to define the \notion{nonsymmetric} Koornwinder polynomials.
  \item Stokman '00: orthogonality of nonsymmetric Koornwinder polynomials.
\end{itemize}
\column{0.33\textwidth}
\centering
\uncover<2->{%
\includegraphics[height=0.4\textheight]{Book cover.jpg}\par\smallskip
Macdonald '03}
\end{columns}
\uncover<3->{\bigskip
$\Longrightarrow$ They generalize several families from symmetric function theory.}


\end{frame}

\begin{frame}{Motivations and previous results}
\begin{itemize}\setlength{\itemsep}{1.2em}
  \item \textbf{Combinatorics.} Type $A$ Macdonald polynomials have several combinatorial formulas: fillings (Haglund--Haiman--Loehr '05) and multiline queues (Corteel--Mandelshtam--Williams '22).\\
  $\blacktriangleright$ Generalize these formulas to Koornwinder polynomials?\\
  \pause
  Alcove walk formulas (Ram--Yip '11, Colmenarejo--Gagnon--Ram '26); the case $\lambda=(1,\dots,1,0,\dots,0)$: rhombic staircase tableaux (Corteel--Mandelshtam--Williams '24).
  \pause
  \item \textbf{Integrability.} Vertex models for nonsymmetric Macdonald polynomials (Borodin--Wheeler '22); only special cases for Koornwinder (Wheeler--Zinn-Justin '16).\\
  $\blacktriangleright$ A vertex model for Koornwinder polynomials?
  \pause
  \item \textbf{Interacting particle systems.} At $q=1$ and $z_i=1$: connection to the multispecies open ASEP (Cantini--Garbali--de Gier--Wheeler '16).
\end{itemize}
\end{frame}

\begin{frame}{Main results (BD--Williams '26)}
\begin{thmA}[two-row recurrence]
Koornwinder polynomials satisfy a recurrence relation.
\end{thmA}
\begin{thmB}[local rules]
The coefficients of this recurrence are characterized by a finite family of local rules.
\end{thmB}
\begin{thmC}[top homogeneous part]
The top homogeneous parts of nonsymmetric Koornwinder polynomials are nonsymmetric type $A$ Macdonald polynomials.
\end{thmC}
\end{frame}

\begin{frame}[t]{Notation}
Fix $n>0$.
\begin{itemize}
  \item A \notion{partition} is $\lambda=(\lambda_1\ge\lambda_2\ge\dots\ge\lambda_n\ge0)$, and a \notion{signed composition} is $\kappa=(\kappa_1,\dots,\kappa_n)\in\ZZ^n$.
  \pause
  \item The finite Weyl group $W_0\simeq S_n\ltimes(\ZZ/2\ZZ)^n$ of type $C_n$ is the group of \notion{signed permutations}. It acts on $\ZZ^n$ by permuting the entries and changing their signs.
  \only<2>{%
  \begin{tikzpicture}[remember picture,overlay,
    dot/.style={circle,inner sep=0pt,draw=blue,fill=white,line width=0.9pt,minimum size=6pt},
    edge/.style={line width=0.5pt}]
    \coordinate (wzero) at (0,0);
    \begin{scope}[shift={($(current page.center|-wzero)+(-3.85cm,-0.75cm)$)},x=1.1cm]
    \draw[edge] (1,0)--(2.45,0);
    \draw[edge,dotted] (2.6,0)--(3.4,0);
    \draw[edge] (3.55,0)--(5,0);
    \draw[edge,double distance=1.6pt] (5,0)--(6,0);
    \draw[edge] (5.577,0.11)--(5.423,0)--(5.577,-0.11);
    \foreach \x/\l in {1/1,2/2,4/n-2,5/n-1,6/n}
      \node[dot,label={[label distance=1pt]below:{\small$\strut\l$}}] at (\x,0) {};
    \end{scope}
  \end{tikzpicture}}
  \pause
  \item $W_0\cdot\lambda$ is the set of signed compositions obtained by permuting the parts of $\lambda$ and changing their signs. For example,
  \[
  W_0\cdot(2,0)=\{(2,0),(-2,0),(0,2),(0,-2)\}.
  \]
  \vspace{-1.2em}
  \pause
  \item We work with Laurent polynomials in $\QQ(q,t,a,b,c,d)[z_1^{\pm1},\dots,z_n^{\pm1}]$. The group $W_0$ acts by permuting the variables and inverting them, $z_i\mapsto z_i^{-1}$.
  \pause
  \item A Laurent polynomial is \notion{symmetric} if it is invariant under $W_0$.
\end{itemize}
\only<6>{%
\begin{exampleblock}{Example}
$z_1^2+z_1^{-2}+z_2^2+z_2^{-2}-bd\,\bigl(z_1z_2+z_1z_2^{-1}+z_1^{-1}z_2+z_1^{-1}z_2^{-1}\bigr)$ is symmetric.
\end{exampleblock}}
\only<7>{%
\medskip
\textbf{Indexing:}
\begin{itemize}
  \item[$\triangleright$] symmetric polynomials: partitions $\lambda=(\lambda_1,\dots,\lambda_n)$,
  \item[$\triangleright$] nonsymmetric Laurent polynomials: signed compositions $\kappa\in\ZZ^n$.
\end{itemize}}
\end{frame}

\begin{frame}{Definitions of $E_\kappa$ and $F_\kappa$}
The affine Hecke algebra of type $\widetilde C_n$ is generated by $T_0,\dots,T_n$, with braid relations given by the diagram and $(T_i-t_i)(T_i+1)=0$.
\begin{center}
\vspace{-0.6em}
\begin{tikzpicture}[x=1.1cm,

    \node[dot,label={[label distance=1pt]below:{\small$\strut\l$}},label={[label distance=1pt]above:{\scriptsize$\important{\p}$}}] at (\x,0) {};
\end{tikzpicture}
\vspace{-0.6em}
\end{center}
\pause
It acts on Laurent polynomials via \important{the polynomial representation}.
\pause
\medskip
\setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}%
\begin{columns}[T,onlytextwidth]
\column{0.415\textwidth}
\small
$E_\kappa$: eigenfunctions of the \notion{Cherednik operators} (Sahi '99, '00)
\begin{align*}
Y_i&=T_i\cdots T_{n-1}T_nT_{n-1}\cdots T_1\\
&\quad T_0T_1^{-1}\cdots T_{i-1}^{-1},
\end{align*}
\[
Y_iE_\kappa=y_i(\kappa)\,E_\kappa,
\]
for $1\le i\le n$, normalized by $[z^\kappa]E_\kappa=1$.
\pause
\column{0.015\textwidth}
\rule{0.4pt}{0.42\textheight}
\column{0.565\textwidth}
\small
$F_\kappa$: the unique family satisfying the \notion{$q$KZ relations} (Kasatani '11), normalized by $[z^\kappa]F_\kappa=1$:
\begin{alignat*}{2}
T_0 F_{\kappa_1,\dots}&=q^{\kappa_1}F_{\important{-\kappa_1},\dots}, &\quad& \kappa_1<0,\\
T_0 F_{\kappa_1,\dots}&=t_0\,F_{\kappa_1,\dots}, && \kappa_1=0,\\
T_i F_{\dots,\kappa_i,\kappa_{i+1},\dots}&=t\,F_{\dots,\kappa_i,\kappa_{i+1},\dots}, && \kappa_i=\kappa_{i+1},\\
T_i F_{\dots,\kappa_i,\kappa_{i+1},\dots}&=F_{\dots,\important{\kappa_{i+1},\kappa_i},\dots}, && \kappa_i>\kappa_{i+1},\\
T_n F_{\dots,\kappa_n}&=t_n\,F_{\dots,\kappa_n}, && \kappa_n=0,\\
T_n F_{\dots,\kappa_n}&=F_{\dots,\important{-\kappa_n}}, && \kappa_n>0.
\end{alignat*}
\end{columns}
\end{frame}

\begin{frame}{Branching rules for symmetric polynomials}
Symmetric Macdonald polynomials:
\[
P_\lambda(z_1,\dots,z_n;q,t)=\sum_{\substack{\mu\subseteq\lambda\\ \lambda/\mu\text{ horizontal strip}}}\important{\psi_{\lambda/\mu}(q,t)}\,z_n^{|\lambda|-|\mu|}P_\mu(z_1,\dots,z_{n-1};q,t),
\]
{\small with branching coefficients (Macdonald '95)
\[
\important{\psi_{\lambda/\mu}(q,t)}=\prod_{s\in R_{\lambda/\mu}\setminus C_{\lambda/\mu}}\frac{b_\mu(s)}{b_\lambda(s)},
\qquad
b_\lambda(s)=\frac{1-q^{a_\lambda(s)}t^{l_\lambda(s)+1}}{1-q^{a_\lambda(s)+1}t^{l_\lambda(s)}},
\]
where $R_{\lambda/\mu}$ (resp.\ $C_{\lambda/\mu}$) is the union of rows (resp.\ columns) meeting $\lambda/\mu$.\par}

\pause
\medskip
Branching rules also exist for
\begin{itemize}
  \item symmetric interpolation Macdonald polynomials (Okounkov '98),
  \item symmetric Koornwinder polynomials (van Diejen--Emsiz '15).
\end{itemize}

\pause
\smallskip
These branching rules have two limitations:
\begin{itemize}
  \item they do not have a direct analogue for nonsymmetric polynomials;
  \item except for the Schur case, the branching coefficients are complicated rational expressions in $q$ and $t$, and the resulting formulas do not make integrality and positivity manifest.
\end{itemize}
\end{frame}

The weight of an MLQ is $\mathrm{wt}=\mathrm{wt}_{\mathrm{ball}}\,\mathrm{wt}_{\mathrm{pairing}}$, where:
\begin{itemize}
  \item Ball weight: each ball in column $i$ contributes $z_i$; here $z_1z_2^2z_3^2z_4z_5z_6^2z_7z_8$.

\end{itemize}
\pause

For $\kappa\in\NN^n$, the type $A$ ASEP polynomial $F^A_\kappa$ is the generating function of MLQs of type $\kappa$.

\begin{frame}{Main proof idea}
\begin{itemize}
  \item Use the double affine Hecke algebra of type $(C^\vee,C)$ (Sahi '99) to define a \important{refinement of the polynomial representation}, in which we have a formal copy $\mathsf Z_1,\dots,\mathsf Z_n$ of the variables $z_1,\dots,z_n$.

  {\small
  \[
  \begin{gathered}
  \widehat F_\kappa=\sum_{\nu}\important{\mathsf Z^\nu}\sum_{\mu}a_\kappa^\mu(\nu)\,F_\mu(z_1,\dots,z_n;q,t,a,b,c,d)\\
  \Big\downarrow{\scriptstyle\,\important{\mathsf Z_i}\mapsto z_i}\\
  F_\kappa(z_1,\dots,z_n;q,t,a,b,c,d)=\sum_{\nu}z^\nu\sum_{\mu}a_\kappa^\mu(\nu)\,F_\mu(z_1,\dots,z_n;q,t,a,b,c,d).
  \end{gathered}
  \]}
\end{itemize}
\end{frame}

\begin{frame}{Two-line configurations}
\[
F_\kappa=\quad\sum_{\mathclap{\substack{\nu\in\ZZ^n\\ \nu_i\in\{0,\Sign(\kappa_i)\}}}}\quad z^\nu\quad\sum_{\mu\in W_0\cdot\kappa^-}a_\kappa^\mu(\nu)\,F_\mu

\begin{itemize}

\end{itemize}

\begin{frame}{Local rules}
\begin{thmB}[BD--Williams '26]
The two-line configurations are characterized by a family of \important{local rules}.
\end{thmB}
Only the columns involved are represented; the other columns are the same on both sides. Here $m>\ell$ and $k>0$.

\medskip
In the bulk (two consecutive columns):
\[
\conf{m & \ell\\ -k & 0}
=t\conf{\ell & m\\ 0 & -k}+(1-t)\conf{m & \ell\\ 0 & -k}.
\]

\medskip
On the boundary (first or last column):
\[
\conf{0 & \cdots\\ \hb{k} & \cdots}
=-q^{k-1}ac\conf{0 & \cdots\\ \hb{-k} & \cdots}-q^{k-1}(a+c)\conf{0 & \cdots\\ -k & \cdots}.
\]
In total, there are $73$ local rules.
\end{frame}

\begin{frame}[t]{A cyclic algorithm}
Each local rule ``transports'' a column to the next position, plus \important{residual terms}.

\only<1-2>{%
\begin{gather*}
\conf{m & \ell\\ -k & 0}
=\underbrace{t\conf{\ell & m\\ 0 & -k}}_{\text{\small transported term}}+\underbrace{(1-t)\conf{m & \ell\\ 0 & -k}}_{\text{\small residual terms}},\\
\conf{0 & \cdots\\ \hb{k} & \cdots}
=\underbrace{-q^{k-1}ac\conf{0 & \cdots\\ \hb{-k} & \cdots}}_{\text{\small transported term}}\underbrace{-q^{k-1}(a+c)\conf{0 & \cdots\\ -k & \cdots}}_{\text{\small residual terms}}.
\end{gather*}
\uncover<2>{\textbf{Fact:} There exists a partial order on two-line configurations for which the residual terms are smaller.}}
\only<3->{%
We solve the system by induction, moving the \act{active column} cyclically:
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{align*}
\CC&=\cconf[3]{\act{2}&0&0\\ \act{0}&1&-2}\\
&\uncover<4->{{}=t\,\cconf[3]{0&\act{2}&0\\ 1&\act{0}&-2}+\text{\small smaller terms}}\\
&\uncover<5->{{}=t\,\cconf[3]{0&0&\act{2}\\ 1&-2&\act{0}}+\text{\small smaller terms}}\\
&\uncover<6->{{}=t\,\cconf[3]{0&0&\act{-2}\\ 1&-2&\act{0}}+\text{\small smaller terms}}
\end{align*}
\end{column}
\begin{column}{0.5\textwidth}
\begin{align*}
&\uncover<7->{{}=t^2\,\cconf[3]{0&\act{-2}&0\\ 1&\act{0}&-2}+\text{\small smaller terms}}\\
&\uncover<8->{{}=t^2\,\cconf[3]{\act{-2}&0&0\\ \act{0}&1&-2}+\text{\small smaller terms}}\\
&\uncover<9->{{}=qt^2\,{\underbrace{\cconf[3]{\act{2}&0&0\\ \act{0}&1&-2}}_{=\,\CC}}+\text{\small smaller terms}}
\end{align*}
\end{column}
\end{columns}
\uncover<10>{%
\vspace{0.6em}
\[
(1-qt^2)\,\CC=\underbrace{(1-t)\cconf[3]{2&0&0\\ 1&0&-2}+t(1-t)\cconf[3]{0&0&-2\\ 1&0&-2}}_{\text{\small smaller terms}}.
\]}
}
\end{frame}

\begin{frame}{Application 1: nonsymmetric Askey--Wilson polynomials}
Theorems A and B give explicit recurrences in the one-variable case ($n=1$):
\setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}%
\begin{theorem}
For $k\ge1$,
\begin{align*}
F_{(k+1)}&=\bigl(z+u_k\bigr)F_{(k)}+\bigl(v_kz+w_k\bigr)F_{(-k)},\\
F_{(-k-1)}&=\bigl(\widetilde v_kz^{-1}+\widetilde w_k\bigr)F_{(k)}+\bigl(z^{-1}+\widetilde u_k\bigr)F_{(-k)},
\end{align*}
with $u_k=w_k\widetilde v_k$, $\widetilde u_k=v_k\widetilde w_k$, and
\begin{alignat*}{2}
v_k&=\qk{k-1}\frac{ac+q-ac\qk{k}-abcd\qk{k}}{1-abcd\qk{2k-1}},&\qquad
w_k&=\qk{k}\frac{-(a+c)+ac(b+d)\qk{k}}{1-abcd\qk{2k}},\\
\widetilde v_k&=\frac{1+bd-bd\qk{k}-abcd\qk{k-1}}{1-abcd\qk{2k-1}},&
\widetilde w_k&=\frac{-(b+d)+bd(a+c)\qk{k}}{1-abcd\qk{2k}}.
\end{alignat*}
\end{theorem}
This result refines the three-term relation for \notion{symmetric} Askey--Wilson polynomials $K_{(k)}=F_{(k)}+F_{(-k)}$.

\pause
\smallskip
All these coefficients are \important{rational functions} of $q^k$. In particular, for $n=1$, all two-line configurations are given by $8$ rational functions of $q^k$. For any $n$, there are only finitely many rational functions to compute.
\end{frame}

\begin{frame}{The affine Hecke algebra $\mathcal H$ of type $\widetilde C_n$}
The affine Hecke algebra $\mathcal H$ is generated by $T_0,T_1,\dots,T_n$, subject to the braid relations
\begin{align*}
T_iT_j&=T_jT_i, && |i-j|\ge 2,\\
T_iT_{i+1}T_i&=T_{i+1}T_iT_{i+1}, && 1\le i<n-1,\\
T_0T_1T_0T_1&=T_1T_0T_1T_0,\\
T_{n-1}T_nT_{n-1}T_n&=T_nT_{n-1}T_nT_{n-1},
\end{align*}
and the quadratic relations
\[
(T_i-t_i)(T_i+1)=0,\qquad 0\le i\le n,
\]
where $\important{t_0=-ac/q}$, $\important{t_i=t}$ for $1\le i<n$, and $\important{t_n=-bd}$.

\begin{frame}{Polynomial representation of $\mathcal H$}
The affine Hecke algebra $\mathcal H$ acts on Laurent polynomials in $z_1,\dots,z_n$ by the Demazure--Lusztig operators
\begin{align*}
T_i &= t + \frac{t z_i-z_{i+1}}{z_i-z_{i+1}}\,(s_i-1),
&& 1\le i<n,\\
T_n &= -bd + \frac{(1-b z_n)(1-d z_n)}{1-z_n^2}\,(s_n-1),\\
T_0 &= -\frac{ac}{q} + \frac{(1-a/z_1)(1-c/z_1)}{1-q/z_1^2}\,(s_0-1).
\end{align*}

\begin{theorem}[Sahi '99]
This representation is faithful.
\end{theorem}

\end{frame}

\begin{frame}{Koornwinder ASEP polynomials}
Fix a partition $\lambda$. The Koornwinder ASEP polynomials $\{F_\eta\}_{\eta\in W_0\cdot\lambda}$ are the unique family of Laurent polynomials, normalized by $[z^\eta]F_\eta=1$, satisfying the $q$KZ relations (Kasatani '11)
\begin{align*}
T_0 F_{\eta_1,\eta_2,\dots,\eta_n}&=q^{\eta_1}F_{\important{-\eta_1},\eta_2,\dots,\eta_n}, && \eta_1<0,\\
T_0 F_{\eta_1,\eta_2,\dots,\eta_n}&=t_0\,F_{\eta_1,\eta_2,\dots,\eta_n}, && \eta_1=0,\\
T_i F_{\dots,\eta_i,\eta_{i+1},\dots}&=t\,F_{\dots,\eta_i,\eta_{i+1},\dots}, && \eta_i=\eta_{i+1},\ 1\le i<n,\\
T_i F_{\dots,\eta_i,\eta_{i+1},\dots}&=F_{\dots,\important{\eta_{i+1},\eta_i},\dots}, && \eta_i>\eta_{i+1},\ 1\le i<n,\\
T_n F_{\dots,\eta_n}&=t_n\,F_{\dots,\eta_n}, && \eta_n=0,\\
T_n F_{\dots,\eta_n}&=F_{\dots,\important{-\eta_n}}, && \eta_n>0.
\end{align*}
Recall that $t_0=-ac/q$ and $t_n=-bd$.
\end{frame}
